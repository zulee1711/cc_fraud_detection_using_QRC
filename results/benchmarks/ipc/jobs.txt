#!/usr/bin/env bash
set -euo pipefail

# Run one plot-backed study from the repository root, for example:
#   bash results/benchmarks/ipc/jobs.txt observables
# Existing logs with an IPC result are skipped. Failed or incomplete logs are
# rerun. Choose one of: precision, depth, window, max-delay, observables,
# spinpulse.

usage() {
    printf 'Usage: bash results/benchmarks/ipc/jobs.txt {precision|depth|window|max-delay|observables|spinpulse}\n' >&2
}

if (($# != 1)); then
    usage
    exit 2
fi

study="$1"

has_ipc_result() {
    python - "$1" <<'PY'
import json
import re
import sys
from pathlib import Path

text = Path(sys.argv[1]).read_text(errors="replace")
marker = re.search(r"Information Processing Capacity\s*\(IPC\)\s*", text)
if marker is None:
    raise SystemExit(1)

try:
    result, _ = json.JSONDecoder().raw_decode(text[marker.end() :])
except json.JSONDecodeError:
    raise SystemExit(1)

if not isinstance(result, dict) or not any(
    key in result for key in ("normalized_total_capacity", "total_capacity")
):
    raise SystemExit(1)
PY
}

run_job() {
    local logfile="$1"
    shift

    mkdir -p "$(dirname "$logfile")"
    if [[ -s "$logfile" ]] && has_ipc_result "$logfile"; then
        printf 'SKIP %s\n' "$logfile"
        return
    fi

    printf 'RUN %s\n' "$logfile"
    if python -m experiments.run_benchmarks --tasks ipc "$@" >"$logfile" 2>&1; then
        if ! has_ipc_result "$logfile"; then
            printf 'ERROR: no valid IPC capacity result was written to %s\n' "$logfile" >&2
            return 1
        fi
    else
        local status=$?
        printf 'ERROR: benchmark failed; see %s\n' "$logfile" >&2
        tail -n 40 "$logfile" >&2
        return "$status"
    fi
}

hamiltonians=(random tfim tfim-longitudinal tfim-native native-exchange)
observables=(Z Z+ZZ XYZ)
seeds_10=($(seq 1 10))

case "$study" in
    precision)
        # Matches precision/plot.py: 50 seeds for each backend and precision.
        for hamiltonian in "${hamiltonians[@]}"; do
            for precision in 0.01 0.05 0.1; do
                for seed in $(seq 1 50); do
                    run_job \
                        "results/benchmarks/ipc/precision/$hamiltonian/shots_precision${precision}_seed${seed}.txt" \
                        --backend shots \
                        --hamiltonian "$hamiltonian" \
                        --precision "$precision" \
                        --seed "$seed"
                done
            done

            # The exact estimator is precision-independent; keep its single
            # plotted point at 0.1 to match the existing result naming.
            for seed in $(seq 1 50); do
                run_job \
                    "results/benchmarks/ipc/precision/$hamiltonian/estimator-exact_precision0.1_seed${seed}.txt" \
                    --backend estimator-exact \
                    --hamiltonian "$hamiltonian" \
                    --precision 0.1 \
                    --seed "$seed"
            done
        done
        ;;

    depth)
        # The depth figure compares these two Hamiltonians and shot results.
        for hamiltonian in random native-exchange; do
            for depth in 1 3 8 15; do
                for seed in "${seeds_10[@]}"; do
                    run_job \
                        "results/benchmarks/ipc/depth/$hamiltonian/shots_precision0.01_depth${depth}_seed${seed}.txt" \
                        --backend shots \
                        --hamiltonian "$hamiltonian" \
                        --precision 0.01 \
                        --ipc-reservoir-depth "$depth" \
                        --seed "$seed"
                done
            done
        done
        ;;

    window)
        # Matches window/plot.py's Hamiltonians, lengths, and filename pattern.
        for hamiltonian in random native-exchange tfim; do
            for window_length in 5 10 15 20; do
                if [[ "$hamiltonian" == random ]]; then
                    depth=4
                else
                    depth=2
                fi
                max_delay=$((window_length - 1))

                for seed in "${seeds_10[@]}"; do
                    run_job \
                        "results/benchmarks/ipc/window/$hamiltonian/shots_precision0.01_window${window_length}_seed${seed}.txt" \
                        --backend shots \
                        --hamiltonian "$hamiltonian" \
                        --precision 0.01 \
                        --ipc-reservoir-depth "$depth" \
                        --ipc-window-length "$window_length" \
                        --max-delay "$max_delay" \
                        --seed "$seed"
                done
            done
        done
        ;;

    max-delay)
        # Delays match the existing max_delay plot grid for each window length.
        for hamiltonian in "${hamiltonians[@]}"; do
            if [[ "$hamiltonian" == random ]]; then
                depth=4
            else
                depth=2
            fi

            for window_length in 5 10 15 20; do
                delays=(1 2 3 4)
                if (( window_length >= 10 )); then
                    delays+=(5)
                fi
                if (( window_length > 5 )); then
                    delays+=("$((window_length - 1))")
                fi

                for backend in shots estimator-exact; do
                    for max_delay in "${delays[@]}"; do
                        for seed in "${seeds_10[@]}"; do
                            run_job \
                                "results/benchmarks/ipc/max_delay/$hamiltonian/${backend}_precision0.01_depth${depth}_window${window_length}_maxdelay${max_delay}_seed${seed}.txt" \
                                --backend "$backend" \
                                --hamiltonian "$hamiltonian" \
                                --precision 0.01 \
                                --ipc-reservoir-depth "$depth" \
                                --ipc-window-length "$window_length" \
                                --max-delay "$max_delay" \
                                --seed "$seed"
                        done
                    done
                done
            done
        done
        ;;

    observables)
        # Classical backends plotted by plot_obs.py.
        for hamiltonian in "${hamiltonians[@]}"; do
            for observable in "${observables[@]}"; do
                for precision in 0.01 0.05 0.1; do
                    for seed in "${seeds_10[@]}"; do
                        run_job \
                            "results/benchmarks/ipc/observables_copy/$hamiltonian/shots_${hamiltonian}_${observable}_precision${precision}_seed${seed}.txt" \
                            --backend shots \
                            --hamiltonian "$hamiltonian" \
                            --observables "$observable" \
                            --precision "$precision" \
                            --seed "$seed"
                    done
                done

                for seed in "${seeds_10[@]}"; do
                    run_job \
                        "results/benchmarks/ipc/observables_copy/$hamiltonian/estimator-exact_${hamiltonian}_${observable}_seed${seed}.txt" \
                        --backend estimator-exact \
                        --hamiltonian "$hamiltonian" \
                        --observables "$observable" \
                        --seed "$seed"
                done
            done
        done
        ;;

    spinpulse)
        # SpinPulse series plotted by plot_obs_sp.py; precision 0.05 is the
        # backend default series, and 0.01 is explicitly recorded in filenames.
        for hamiltonian in "${hamiltonians[@]}"; do
            if [[ "$hamiltonian" == random ]]; then
                depth=4
            else
                depth=2
            fi

            for observable in Z Z+ZZ; do
                for precision in 0.05 0.01; do
                    for seed in "${seeds_10[@]}"; do
                        if [[ "$precision" == 0.05 ]]; then
                            logfile="results/benchmarks/ipc/observables_copy/$hamiltonian/spinpulse_${hamiltonian}_${observable}_seed${seed}.txt"
                        else
                            logfile="results/benchmarks/ipc/observables_copy/$hamiltonian/spinpulse_${hamiltonian}_${observable}_precision_0.01_seed${seed}.txt"
                        fi
                        run_job "$logfile" \
                            --backend spinpulse \
                            --hardware realistic \
                            --noise pink \
                            --t2-us 10 \
                            --tj-us 10 \
                            --time-step-ns 1 \
                            --noise-samples 20 \
                            --ipc-length 3000 \
                            --ipc-window-length 15 \
                            --ipc-reservoir-depth "$depth" \
                            --max-degree 3 \
                            --max-delay 10 \
                            --hamiltonian "$hamiltonian" \
                            --observables "$observable" \
                            --precision "$precision" \
                            --seed "$seed"
                    done
                done
            done
        done
        ;;

    *)
        printf 'Unknown study: %s\n' "$study" >&2
        usage
        exit 2
        ;;
esac
