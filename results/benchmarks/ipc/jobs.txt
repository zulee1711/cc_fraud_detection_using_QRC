# 1. Precision 

mkdir -p results/benchmarks/ipc/precision

# Add more hamiltonians/change param range if you want
for hamiltonian in random tfim tfim-longitudinal tfim-native native-exchange; do
    mkdir -p "results/benchmarks/ipc/precision/${hamiltonian}"

    for backend in shots; do
        for precision in 0.01 0.05 0.1; do
            for seed in $(seq 1 10); do
                echo "Running hamiltonian=$hamiltonian backend=$backend precision=$precision seed=$seed"

                python -m experiments.run_benchmarks \
                    --tasks ipc \
                    --backend "$backend" \
                    --hamiltonian "$hamiltonian" \
                    --precision "$precision" \
                    --seed "$seed" \
                    > "results/benchmarks/ipc/precision/${hamiltonian}/${backend}_precision${precision}_seed${seed}.txt" 2>&1
            done
        done
    done

    for backend in estimator-exact; do

            for seed in $(seq 1 50); do
                echo "Running hamiltonian=$hamiltonian backend=$backend precision=$precision seed=$seed"

                python -m experiments.run_benchmarks \
                    --tasks ipc \
                    --backend "$backend" \
                    --hamiltonian "$hamiltonian" \
                    --precision "$precision" \
                    --seed "$seed" \
                    > "results/benchmarks/ipc/precision/${hamiltonian}/${backend}_precision${precision}_seed${seed}.txt" 2>&1
            done

    done
done

# 2. Depth

mkdir -p results/benchmarks/ipc/depth

for hamiltonian in random tfim tfim-longitudinal tfim-native native-exchange; do
    mkdir -p "results/benchmarks/ipc/depth/${hamiltonian}"
    for backend in shots estimator-exact; do
        for depth in 1 3 8 15; do
            seq 1 10 | xargs -P 1 -I {} bash -c '
                hamiltonian="$1"
                depth="$2"
                seed="$3"

                echo "Running hamiltonian=$hamiltonian depth=$depth backend=shots precision=0.01 seed=$seed"

                python -m experiments.run_benchmarks \
                    --tasks ipc \
                    --backend "$backend" \
                    --hamiltonian "$hamiltonian" \
                    --precision 0.01 \
                    --seed "$seed" \
                    --ipc-reservoir-depth "$depth" \
                    > "results/benchmarks/ipc/depth/${hamiltonian}/shots_precision0.01_depth${depth}_seed${seed}.txt" 2>&1
            ' _ "$hamiltonian" "$depth" {}
        done
    done
done

for hamiltonian in tfim-longitudinal tfim-native native-exchange; do
    out="results/benchmarks/ipc/observables/${hamiltonian}"
    mkdir -p "$out"

    for backend in shots estimator-exact; do
        for precision in 0.01 0.05 0.1; do
            if [ "$backend" = "estimator-exact" ] && [ "$precision" != "0.01" ]; then
                continue
            fi

            for obs in Z Z+ZZ XYZ XYZ+XYZ-2body XYZ+XYZ-2body-all; do
                for seed in $(seq 1 10); do
                    logfile="${out}/${backend}_obs-${obs}_precision${precision}_seed${seed}.txt"

                    echo "Running hamiltonian=$hamiltonian observables=$obs backend=$backend precision=$precision seed=$seed"

                    python -m experiments.run_benchmarks \
                        --tasks ipc \
                        --backend "$backend" \
                        --hamiltonian "$hamiltonian" \
                        --precision "$precision" \
                        --observables "$obs" \
                        --seed "$seed" \
                        > "$logfile" 2>&1

                    if [ $? -ne 0 ]; then
                        echo "FAILED: $logfile"
                        tail -20 "$logfile"
                    fi
                done
            done
        done
    done
done



#!/usr/bin/env bash
set -euo pipefail

OUT="results/benchmarks/ipc/observables"
mkdir -p "$OUT"

HAMILTONIANS=(
    random
    tfim
    tfim-longitudinal
    tfim-native
    native-exchange
)

OBSERVABLES=(
    Z
    Z+ZZ
    XYZ
    XYZ+XYZ-2body
    XYZ+XYZ-2body-all
)

for hamiltonian in "${HAMILTONIANS[@]}"; do
    for observables in "${OBSERVABLES[@]}"; do
        for seed in $(seq 1 10); do
            safe_obs="${observables//+/_}"

            echo "Hamiltonian=$hamiltonian observables=$observables seed=$seed"

            python -m experiments.run_benchmarks \
                --tasks ipc \
                --backend spinpulse \
                --hardware realistic \
                --noise pink \
                --t2-us 0.42 \
                --tj-us 0.42 \
                --time-step-ns 1 \
                --noise-samples 20 \
                --hamiltonian "$hamiltonian" \
                --observables "$observables" \
                --seed "$seed" \
                --ipc-length 3000 \
                --ipc-window-length 15 \
                --max-degree 3 \
                --max-delay 10 \
                > "$OUT/${hamiltonian}/$spinpulse__${safe_obs}_seed${seed}.txt" 2>&1
        done
    done
done
